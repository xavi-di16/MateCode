// ============================================================================
// UBICACIÓN: api/send-summary.ts
// ============================================================================
import { SESClient, SendEmailCommand } from '@aws-sdk/client-ses';
import type { VercelRequest, VercelResponse } from '@vercel/node';

interface SummaryTask {
    title: string;
    completed: boolean;
}

interface RequestBody {
    toEmail?: unknown;
    userName?: unknown;
    tasks?: unknown;
}

function escapeHtml(text: string): string {
    return text
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;');
}

function isValidTaskArray(value: unknown): value is SummaryTask[] {
    return (
        Array.isArray(value) &&
        value.every(
            (item) =>
                typeof item === 'object' &&
                item !== null &&
                typeof (item as SummaryTask).title === 'string' &&
                typeof (item as SummaryTask).completed === 'boolean'
        )
    );
}

export default async function handler(req: VercelRequest, res: VercelResponse) {
    if (req.method !== 'POST') {
        return res.status(405).json({ message: 'Método no permitido. Usá POST.' });
    }

    const region = process.env.AWS_REGION;
    const accessKeyId = process.env.AWS_ACCESS_KEY_ID;
    const secretAccessKey = process.env.AWS_SECRET_ACCESS_KEY;
    const fromEmail = process.env.AWS_SES_FROM_EMAIL;

    if (!region || !accessKeyId || !secretAccessKey || !fromEmail) {
        return res.status(500).json({
            message: 'Faltan variables de entorno de AWS SES en el servidor.',
        });
    }

    const body = (req.body ?? {}) as RequestBody;
    const toEmail = typeof body.toEmail === 'string' ? body.toEmail.trim() : '';
    const userName =
        typeof body.userName === 'string' && body.userName.trim()
            ? body.userName.trim()
            : 'Usuario';

    if (!toEmail || !toEmail.includes('@')) {
        return res.status(400).json({ message: 'El correo de destino no es válido.' });
    }

    if (!isValidTaskArray(body.tasks) || body.tasks.length === 0) {
        return res.status(400).json({
            message: 'Debés enviar al menos una tarea válida para generar el resumen.',
        });
    }

    const completedTasks = body.tasks.filter((t) => t.completed);
    const pendingTasks = body.tasks.filter((t) => !t.completed);

    const htmlList = body.tasks
        .map(
            (t) =>
                `<li><strong>${t.completed ? '[Completada]' : '[Pendiente]'}</strong> ${escapeHtml(t.title)}</li>`
        )
        .join('');

    const textList = body.tasks
        .map((t) => `- ${t.completed ? '[Completada]' : '[Pendiente]'} ${t.title}`)
        .join('\n');

    const sesClient = new SESClient({
        region,
        credentials: { accessKeyId, secretAccessKey },
    });

    const command = new SendEmailCommand({
        Source: fromEmail,
        Destination: { ToAddresses: [toEmail] },
        Message: {
            Subject: {
                Data: `Resumen de tareas - MateCode (${pendingTasks.length} pendientes)`,
                Charset: 'UTF-8',
            },
            Body: {
                Text: {
                    Data: `Hola ${userName},\n\nResumen de tus tareas:\nTotal: ${body.tasks.length} | Completadas: ${completedTasks.length} | Pendientes: ${pendingTasks.length}\n\n${textList}`,
                    Charset: 'UTF-8',
                },
                Html: {
                    Data: `
            <h2>Resumen de tus tareas - MateCode</h2>
            <p>Hola <strong>${escapeHtml(userName)}</strong>,</p>
            <p>Total: <strong>${body.tasks.length}</strong> · Completadas: <strong>${completedTasks.length}</strong> · Pendientes: <strong>${pendingTasks.length}</strong></p>
            <ul>${htmlList}</ul>
          `,
                    Charset: 'UTF-8',
                },
            },
        },
    });

    try {
        const result = await sesClient.send(command);
        return res.status(200).json({
            message: 'Resumen enviado por email correctamente.',
            messageId: result.MessageId,
        });
    } catch (error) {
        const errorMessage =
            error instanceof Error ? error.message : 'Error desconocido al enviar el correo.';
        return res.status(500).json({
            message: `No se pudo enviar el email con AWS SES: ${errorMessage}`,
        });
    }
}

// ============================================================================
// UBICACIÓN: firestore.rules
// ============================================================================
rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    match /tasks/{taskId} {
      // Solo el dueño autenticado puede leer o eliminar sus tareas
      allow read, delete: if request.auth != null
        && resource.data.userId == request.auth.uid;

      // Al crear, el userId debe coincidir con el usuario autenticado y validar tipos básicos
      allow create: if request.auth != null
        && request.resource.data.userId == request.auth.uid
        && request.resource.data.title is string
        && request.resource.data.title.size() > 0
        && request.resource.data.description is string
        && request.resource.data.completed is bool;

      // Al actualizar, el dueño no puede reasignar la tarea a otro userId
      allow update: if request.auth != null
        && resource.data.userId == request.auth.uid
        && request.resource.data.userId == resource.data.userId
        && request.resource.data.title is string
        && request.resource.data.title.size() > 0
        && request.resource.data.description is string
        && request.resource.data.completed is bool;
    }
  }
}

import { BrowserRouter, Navigate, Route, Routes } from 'react-router-dom';
import { AuthProvider } from './features/auth/AuthContext';
import { LoginPage } from './pages/LoginPage';
import { RegisterPage } from './pages/RegisterPage';
import { TasksPage } from './pages/TasksPage';
import { ProtectedRoute } from './routes/ProtectedRoute';
import { PublicRoute } from './routes/PublicRoute';

export function App() {
  return (
    <AuthProvider>
      <BrowserRouter>
        <Routes>
          <Route
            path="/login"
            element={
              <PublicRoute>
                <LoginPage />
              </PublicRoute>
            }
          />
          <Route
            path="/register"
            element={
              <PublicRoute>
                <RegisterPage />
              </PublicRoute>
            }
          />
          <Route
            path="/tasks"
            element={
              <ProtectedRoute>
                <TasksPage />
              </ProtectedRoute>
            }
          />
          <Route path="*" element={<Navigate to="/tasks" replace />} />
        </Routes>
      </BrowserRouter>
    </AuthProvider>
  );
}

export default App;

import { useState, type FormEvent } from 'react';
import type { TaskFormValues } from '../types/task.types';

interface TodoFormProps {
    onSubmit: (values: TaskFormValues) => Promise<void> | void;
}

export function TodoForm({ onSubmit }: TodoFormProps) {
    const [title, setTitle] = useState('');
    const [description, setDescription] = useState('');
    const [validationError, setValidationError] = useState<string | null>(null);
    const [submitting, setSubmitting] = useState(false);

    const handleSubmit = async (event: FormEvent<HTMLFormElement>) => {
        event.preventDefault();

        if (!title.trim()) {
            setValidationError('El título es obligatorio.');
            return;
        }

        setValidationError(null);
        setSubmitting(true);
        try {
            await onSubmit({ title: title.trim(), description: description.trim() });
            setTitle('');
            setDescription('');
        } catch {
            // El error de red/Firestore se muestra en el contenedor padre
        } finally {
            setSubmitting(false);
        }
    };

    return (
        <form onSubmit={handleSubmit} className="todo-form" noValidate>
            <h2>Nueva tarea</h2>

            {validationError && (
                <p className="alert-error" role="alert">
                    {validationError}
                </p>
            )}

            <div className="todo-form-fields">
                <label className="field">
                    <span>Título</span>
                    <input
                        type="text"
                        value={title}
                        onChange={(e) => setTitle(e.target.value)}
                        placeholder="Ej: Revisar facturas del mes"
                        maxLength={100}
                        required
                    />
                </label>

                <label className="field">
                    <span>Descripción</span>
                    <textarea
                        value={description}
                        onChange={(e) => setDescription(e.target.value)}
                        placeholder="Detalles opcionales de la tarea..."
                        rows={2}
                        maxLength={300}
                    />
                </label>
            </div>

            <button type="submit" className="btn-primary" disabled={submitting}>
                {submitting ? 'Guardando...' : 'Agregar tarea'}
            </button>
        </form>
    );
}

import { useState, type FormEvent } from 'react';
import type { Task, TaskFormValues } from '../types/task.types';

interface TodoItemProps {
    task: Task;
    onToggle: (task: Task) => Promise<void> | void;
    onEdit: (taskId: string, values: TaskFormValues) => Promise<void> | void;
    onDelete: (taskId: string) => Promise<void> | void;
}

export function TodoItem({ task, onToggle, onEdit, onDelete }: TodoItemProps) {
    const [isEditing, setIsEditing] = useState(false);
    const [editTitle, setEditTitle] = useState(task.title);
    const [editDescription, setEditDescription] = useState(task.description);
    const [editError, setEditError] = useState<string | null>(null);
    const [busy, setBusy] = useState(false);

    const handleSaveEdit = async (event: FormEvent<HTMLFormElement>) => {
        event.preventDefault();
        if (!editTitle.trim()) {
            setEditError('El título no puede quedar vacío.');
            return;
        }

        setEditError(null);
        setBusy(true);
        try {
            await onEdit(task.id, {
                title: editTitle.trim(),
                description: editDescription.trim(),
            });
            setIsEditing(false);
        } finally {
            setBusy(false);
        }
    };

    const handleCancelEdit = () => {
        setEditTitle(task.title);
        setEditDescription(task.description);
        setEditError(null);
        setIsEditing(false);
    };

    if (isEditing) {
        return (
            <li className="todo-item todo-item--editing">
                <form onSubmit={handleSaveEdit} className="todo-edit-form" noValidate>
                    {editError && (
                        <p className="alert-error" role="alert">
                            {editError}
                        </p>
                    )}
                    <input
                        type="text"
                        aria-label="Editar título"
                        value={editTitle}
                        onChange={(e) => setEditTitle(e.target.value)}
                    />
                    <textarea
                        aria-label="Editar descripción"
                        value={editDescription}
                        onChange={(e) => setEditDescription(e.target.value)}
                        rows={2}
                    />
                    <div className="todo-actions">
                        <button type="submit" className="btn-primary btn-sm" disabled={busy}>
                            Guardar
                        </button>
                        <button
                            type="button"
                            className="btn-secondary btn-sm"
                            onClick={handleCancelEdit}
                            disabled={busy}
                        >
                            Cancelar
                        </button>
                    </div>
                </form>
            </li>
        );
    }

    return (
        <li className={`todo-item ${task.completed ? 'todo-item--completed' : ''}`}>
            <div className="todo-main">
                <input
                    type="checkbox"
                    checked={task.completed}
                    onChange={() => onToggle(task)}
                    aria-label={`Marcar "${task.title}" como ${task.completed ? 'pendiente' : 'completada'}`}
                />
                <div className="todo-content">
                    <h3 className="todo-title">{task.title}</h3>
                    {task.description && <p className="todo-description">{task.description}</p>}
                </div>
            </div>

            <div className="todo-actions">
                <button
                    type="button"
                    className="btn-secondary btn-sm"
                    onClick={() => setIsEditing(true)}
                >
                    Editar
                </button>
                <button
                    type="button"
                    className="btn-danger btn-sm"
                    onClick={() => onDelete(task.id)}
                >
                    Eliminar
                </button>
            </div>
        </li>
    );
}

import { createContext, useCallback, useEffect, useState, type ReactNode } from 'react';
import type { AuthContextValue, AuthFormValues, AuthUser } from '../../types/auth.types';
import { getAuthErrorMessage } from '../../utils/firebaseErrors';
import {
    loginWithEmail,
    loginWithGooglePopup,
    logoutUser,
    registerWithEmail,
    subscribeToAuthChanges,
} from './authService';

export const AuthContext = createContext<AuthContextValue | null>(null);

interface AuthProviderProps {
    children: ReactNode;
}

export function AuthProvider({ children }: AuthProviderProps) {
    const [user, setUser] = useState<AuthUser | null>(null);
    const [loading, setLoading] = useState<boolean>(true);
    const [error, setError] = useState<string | null>(null);

    useEffect(() => {
        const unsubscribe = subscribeToAuthChanges((currentUser) => {
            setUser(currentUser);
            setLoading(false);
        });

        return () => unsubscribe();
    }, []);

    const clearError = useCallback(() => {
        setError(null);
    }, []);

    const register = useCallback(async (values: AuthFormValues) => {
        setError(null);
        try {
            await registerWithEmail(values);
        } catch (err) {
            setError(getAuthErrorMessage(err));
            throw err;
        }
    }, []);

    const login = useCallback(async (values: AuthFormValues) => {
        setError(null);
        try {
            await loginWithEmail(values);
        } catch (err) {
            setError(getAuthErrorMessage(err));
            throw err;
        }
    }, []);

    const loginWithGoogle = useCallback(async () => {
        setError(null);
        try {
            await loginWithGooglePopup();
        } catch (err) {
            setError(getAuthErrorMessage(err));
            throw err;
        }
    }, []);

    const logout = useCallback(async () => {
        setError(null);
        try {
            await logoutUser();
        } catch (err) {
            setError(getAuthErrorMessage(err));
            throw err;
        }
    }, []);

    return (
        <AuthContext.Provider
            value={{ user, loading, error, register, login, loginWithGoogle, logout, clearError }}
        >
            {children}
        </AuthContext.Provider>
    );
}

import {
    createUserWithEmailAndPassword,
    GoogleAuthProvider,
    onAuthStateChanged,
    signInWithEmailAndPassword,
    signInWithPopup,
    signOut,
    type Unsubscribe,
    type User,
} from 'firebase/auth';
import { auth } from '../../services/firebase';
import type { AuthFormValues, AuthUser } from '../../types/auth.types';

const googleProvider = new GoogleAuthProvider();

function mapFirebaseUser(user: User | null): AuthUser | null {
    if (!user) return null;
    return {
        uid: user.uid,
        email: user.email,
        displayName: user.displayName,
    };
}

export async function registerWithEmail({ email, password }: AuthFormValues): Promise<AuthUser> {
    const credential = await createUserWithEmailAndPassword(auth, email.trim(), password);
    return {
        uid: credential.user.uid,
        email: credential.user.email,
        displayName: credential.user.displayName,
    };
}

export async function loginWithEmail({ email, password }: AuthFormValues): Promise<AuthUser> {
    const credential = await signInWithEmailAndPassword(auth, email.trim(), password);
    return {
        uid: credential.user.uid,
        email: credential.user.email,
        displayName: credential.user.displayName,
    };
}

export async function loginWithGooglePopup(): Promise<AuthUser> {
    const credential = await signInWithPopup(auth, googleProvider);
    return {
        uid: credential.user.uid,
        email: credential.user.email,
        displayName: credential.user.displayName,
    };
}

export async function logoutUser(): Promise<void> {
    await signOut(auth);
}

export function subscribeToAuthChanges(callback: (user: AuthUser | null) => void): Unsubscribe {
    return onAuthStateChanged(auth, (firebaseUser) => {
        callback(mapFirebaseUser(firebaseUser));
    });
}

import {
    addDoc,
    collection,
    deleteDoc,
    doc,
    onSnapshot,
    query,
    updateDoc,
    where,
    type Unsubscribe,
} from 'firebase/firestore';
import { db } from '../../services/firebase';
import type { Task, TaskFormValues } from '../../types/task.types';

const TASKS_COLLECTION = 'tasks';

export function subscribeToUserTasks(
    userId: string,
    onData: (tasks: Task[]) => void,
    onError: (error: Error) => void
): Unsubscribe {
    const tasksRef = collection(db, TASKS_COLLECTION);
    const userTasksQuery = query(tasksRef, where('userId', '==', userId));

    return onSnapshot(
        userTasksQuery,
        (snapshot) => {
            const tasks: Task[] = snapshot.docs.map((document) => {
                const data = document.data();
                return {
                    id: document.id,
                    userId: String(data.userId ?? ''),
                    title: String(data.title ?? ''),
                    description: String(data.description ?? ''),
                    completed: Boolean(data.completed),
                    createdAt: Number(data.createdAt ?? 0),
                    updatedAt: Number(data.updatedAt ?? 0),
                };
            });

            tasks.sort((a, b) => b.createdAt - a.createdAt);
            onData(tasks);
        },
        (error) => {
            onError(error);
        }
    );
}

export async function createTask(userId: string, values: TaskFormValues): Promise<void> {
    const trimmedTitle = values.title.trim();
    if (!trimmedTitle) {
        throw new Error('El título de la tarea no puede estar vacío.');
    }

    const now = Date.now();
    await addDoc(collection(db, TASKS_COLLECTION), {
        userId,
        title: trimmedTitle,
        description: values.description.trim(),
        completed: false,
        createdAt: now,
        updatedAt: now,
    });
}

export async function updateTask(
    taskId: string,
    updates: Partial<Pick<Task, 'title' | 'description' | 'completed'>>
): Promise<void> {
    const taskRef = doc(db, TASKS_COLLECTION, taskId);
    const payload: Record<string, string | boolean | number> = {
        updatedAt: Date.now(),
    };

    if (updates.title !== undefined) {
        const trimmedTitle = updates.title.trim();
        if (!trimmedTitle) {
            throw new Error('El título de la tarea no puede estar vacío.');
        }
        payload.title = trimmedTitle;
    }

    if (updates.description !== undefined) {
        payload.description = updates.description.trim();
    }

    if (updates.completed !== undefined) {
        payload.completed = updates.completed;
    }

    await updateDoc(taskRef, payload);
}

export async function deleteTask(taskId: string): Promise<void> {
    const taskRef = doc(db, TASKS_COLLECTION, taskId);
    await deleteDoc(taskRef);
}

import { useCallback, useEffect, useMemo, useState } from 'react';
import {
    createTask,
    deleteTask,
    subscribeToUserTasks,
    updateTask,
} from '../features/tasks/taskService';
import type { Task, TaskFilter, TaskFormValues } from '../types/task.types';

export function useTasks(userId: string | undefined) {
    const [tasks, setTasks] = useState<Task[]>([]);
    const [loading, setLoading] = useState<boolean>(true);
    const [error, setError] = useState<string | null>(null);
    const [filter, setFilter] = useState<TaskFilter>('all');

    useEffect(() => {
        if (!userId) {
            setTasks([]);
            setLoading(false);
            return;
        }

        setLoading(true);
        setError(null);

        const unsubscribe = subscribeToUserTasks(
            userId,
            (incomingTasks) => {
                setTasks(incomingTasks);
                setLoading(false);
            },
            () => {
                setError('No se pudieron cargar tus tareas. Verificá tu conexión o permisos.');
                setLoading(false);
            }
        );

        return () => unsubscribe();
    }, [userId]);

    const addTask = useCallback(
        async (values: TaskFormValues) => {
            if (!userId) return;
            setError(null);
            try {
                await createTask(userId, values);
            } catch (err) {
                const message = err instanceof Error ? err.message : 'No se pudo crear la tarea.';
                setError(message);
                throw err;
            }
        },
        [userId]
    );

    const editTask = useCallback(async (taskId: string, values: TaskFormValues) => {
        setError(null);
        try {
            await updateTask(taskId, values);
        } catch (err) {
            const message = err instanceof Error ? err.message : 'No se pudo actualizar la tarea.';
            setError(message);
            throw err;
        }
    }, []);

    const toggleTask = useCallback(async (task: Task) => {
        setError(null);
        try {
            await updateTask(task.id, { completed: !task.completed });
        } catch (err) {
            const message = err instanceof Error ? err.message : 'No se pudo cambiar el estado.';
            setError(message);
            throw err;
        }
    }, []);

    const removeTask = useCallback(async (taskId: string) => {
        setError(null);
        try {
            await deleteTask(taskId);
        } catch (err) {
            const message = err instanceof Error ? err.message : 'No se pudo eliminar la tarea.';
            setError(message);
            throw err;
        }
    }, []);

    const filteredTasks = useMemo(() => {
        if (filter === 'pending') return tasks.filter((t) => !t.completed);
        if (filter === 'completed') return tasks.filter((t) => t.completed);
        return tasks;
    }, [tasks, filter]);

    const counts = useMemo(
        () => ({
            all: tasks.length,
            pending: tasks.filter((t) => !t.completed).length,
            completed: tasks.filter((t) => t.completed).length,
        }),
        [tasks]
    );

    return {
        tasks,
        filteredTasks,
        counts,
        filter,
        setFilter,
        loading,
        error,
        addTask,
        editTask,
        toggleTask,
        removeTask,
    };
}

import { useState } from 'react';
import { TodoForm } from '../components/TodoForm';
import { TodoList } from '../components/TodoList';
import { useAuth } from '../hooks/useAuth';
import { useTasks } from '../hooks/useTasks';
import { sendTasksSummaryEmail } from '../services/emailService';
import type { TaskFilter } from '../types/task.types';

const FILTER_LABELS: { key: TaskFilter; label: string }[] = [
    { key: 'all', label: 'Todas' },
    { key: 'pending', label: 'Pendientes' },
    { key: 'completed', label: 'Completadas' },
];

type EmailStatus = 'idle' | 'loading' | 'success' | 'error';

export function TasksPage() {
    const { user, logout } = useAuth();
    const {
        tasks,
        filteredTasks,
        counts,
        filter,
        setFilter,
        loading,
        error,
        addTask,
        editTask,
        toggleTask,
        removeTask,
    } = useTasks(user?.uid);

    const [emailStatus, setEmailStatus] = useState<EmailStatus>('idle');
    const [emailMessage, setEmailMessage] = useState<string | null>(null);

    const handleSendSummary = async () => {
        if (!user?.email) return;

        setEmailStatus('loading');
        setEmailMessage(null);

        try {
            const response = await sendTasksSummaryEmail({
                toEmail: user.email,
                userName: user.displayName || user.email,
                tasks: tasks.map(({ title, completed }) => ({ title, completed })),
            });
            setEmailStatus('success');
            setEmailMessage(response.message);
        } catch (err) {
            setEmailStatus('error');
            setEmailMessage(
                err instanceof Error ? err.message : 'No se pudo enviar el resumen por email.'
            );
        }
    };

    return (
        <div className="app-shell">
            <header className="app-header">
                <div>
                    <h1>MateCode Tareas</h1>
                    <p className="user-badge">{user?.displayName || user?.email}</p>
                </div>

                <div className="header-actions">
                    <button
                        type="button"
                        className="btn-primary btn-sm"
                        onClick={handleSendSummary}
                        disabled={emailStatus === 'loading' || tasks.length === 0}
                    >
                        {emailStatus === 'loading' ? 'Enviando...' : 'Enviar resumen por email'}
                    </button>
                    <button type="button" className="btn-secondary btn-sm" onClick={logout}>
                        Cerrar sesión
                    </button>
                </div>
            </header>

            <main className="tasks-main">
                {error && (
                    <div className="alert-error" role="alert">
                        {error}
                    </div>
                )}

                {emailStatus === 'success' && emailMessage && (
                    <div className="alert-success" role="status">
                        {emailMessage}
                    </div>
                )}

                {emailStatus === 'error' && emailMessage && (
                    <div className="alert-error" role="alert">
                        {emailMessage}
                    </div>
                )}

                <TodoForm onSubmit={addTask} />

                <section className="tasks-panel" aria-label="Listado de tareas">
                    <div className="filter-bar" role="tablist" aria-label="Filtrar tareas">
                        {FILTER_LABELS.map(({ key, label }) => (
                            <button
                                key={key}
                                type="button"
                                role="tab"
                                aria-selected={filter === key}
                                className={`filter-btn ${filter === key ? 'filter-btn--active' : ''}`}
                                onClick={() => setFilter(key)}
                            >
                                {label} ({counts[key]})
                            </button>
                        ))}
                    </div>

                    <TodoList
                        tasks={filteredTasks}
                        loading={loading}
                        onToggle={toggleTask}
                        onEdit={editTask}
                        onDelete={removeTask}
                    />
                </section>
            </main>
        </div>
    );
}

    children: ReactNode;
}

export function ProtectedRoute({ children }: ProtectedRouteProps) {
    const { user, loading } = useAuth();

    if (loading) {
        return (
            <div className="screen-center" role="status" aria-live="polite">
                <p>Verificando sesión...</p>

    if (!user) {
        return <Navigate to="/login" replace />;

    return <>{children}</>;

import type { SendSummaryPayload, SendSummaryResponse } from '../types/task.types';

export async function sendTasksSummaryEmail(
    payload: SendSummaryPayload
): Promise<SendSummaryResponse> {
    if (!payload.toEmail || !payload.toEmail.includes('@')) {
        throw new Error('No se encontró un correo electrónico válido para el usuario.');
    }

    if (payload.tasks.length === 0) {
        throw new Error('Agregá al menos una tarea antes de enviar el resumen.');
    }

    const response = await fetch('/api/send-summary', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload),
    });

    const data = (await response.json().catch(() => ({}))) as Partial<SendSummaryResponse>;

    if (!response.ok) {
        throw new Error(data.message ?? 'Error al comunicarse con el servidor de correos.');
    }

    return {
        message: data.message ?? 'Resumen enviado correctamente.',
        messageId: data.messageId,
    };
}

import { FirebaseError } from 'firebase/app';

const AUTH_ERROR_MESSAGES: Record<string, string> = {
    'auth/invalid-credential': 'El correo o la contraseña son incorrectos.',
    'auth/user-not-found': 'No existe una cuenta registrada con este correo.',
    'auth/wrong-password': 'La contraseña ingresada es incorrecta.',
    'auth/email-already-in-use': 'Ya existe una cuenta registrada con este correo.',
    'auth/invalid-email': 'El formato del correo electrónico no es válido.',
    'auth/weak-password': 'La contraseña debe tener al menos 6 caracteres.',
    'auth/too-many-requests': 'Demasiados intentos fallidos. Intentá de nuevo más tarde.',
    'auth/popup-closed-by-user': 'Cerraste la ventana de inicio de sesión con Google antes de completar el proceso.',
    'auth/network-request-failed': 'Error de red. Verificá tu conexión a internet.',
};

export function getAuthErrorMessage(error: unknown): string {
    if (error instanceof FirebaseError) {
        return AUTH_ERROR_MESSAGES[error.code] ?? 'Ocurrió un error al autenticar. Intentá nuevamente.';
    }
    if (error instanceof Error) {
        return error.message;
    }
    return 'Ocurrió un error inesperado.';
}
